get_palette <- function(name, n) {
  base <- if (identical(name, "gray")) gray.colors(max(n, 1), start = 0.2, end = 0.7)
          else if (!is.null(PALETTES[[name]])) PALETTES[[name]]
          else PALETTES[[DEFAULT_PALETTE]]
  rep_len(adjustcolor(base, alpha.f = 0.8), n)
}

# ── Colonne di metadati ───────────────────────────────────────────────────────
META_COLS     <- c("unit","timestamp","sensor","genotype","g_alias","treatment",
                   "date","posA","posX","posY")
ID_COLS_DAILY <- c("unit","date","treatment","posX","posY","genotype","n_scans")
# Etichetta per i vasi senza trattamento: "" non può fare da nome di un colore
NO_TREATMENT  <- "(none)"
# Unità riconosciute nell'ultima parola del nome di un parametro
UNIT_TOKENS   <- c("nm","mm","cm","dm","m","mm2","cm2","dm2","m2","mm3","cm3","dm3","m3",
                   "mm\u00b2","cm\u00b2","dm\u00b2","m\u00b2","mm\u00b3","cm\u00b3","dm\u00b3","m\u00b3",
                   "g","mg","kg","\u00b5g","\u00b0","deg","px")
param_unit <- function(p) {
  p   <- trimws(p)
  tok <- gsub("^\\(|\\)$", "", sub("^.*\\s", "", p))
  if (grepl("\\s", p) && tok %in% UNIT_TOKENS) tok else ""
}

# ── Date: ISO o formati europei, anche con l'ora in coda ─────────────────────
parse_pe_date <- function(x) {
  s  <- trimws(as.character(x)); ok <- !is.na(s) & nzchar(s)
  fmts <- c("%Y-%m-%d","%Y/%m/%d","%d/%m/%Y","%d.%m.%Y","%d-%m-%Y","%m/%d/%Y")
  read_fmt <- function(f) {
    d <- as.Date(rep(NA, length(s)))
    d[ok] <- as.Date(s[ok], format = f)
    y <- as.integer(format(d, "%Y"))
    d[!is.na(y) & (y < 1990 | y > 2100)] <- NA
    d
  }
  # Vale il formato che legge più righe: una data rovinata non scarta tutto il file
  parsed <- lapply(fmts, read_fmt)
  n_ok   <- vapply(parsed, function(d) sum(!is.na(d)), 0L)
  best   <- which.max(n_ok)
  list(date = parsed[[best]],
       ambiguous = fmts[best] == "%d/%m/%Y" && n_ok[best] > 0 &&
                   n_ok[fmts == "%m/%d/%Y"] == n_ok[best])
}

# ── Parsing PlantEye ──────────────────────────────────────────────────────────
parse_planteye <- function(filepath) {
  # Tutto come testo: read.csv altrimenti trasforma i metadati (T/F -> TRUE/FALSE,
  # 01 -> 1, NA -> mancante). I parametri diventano numeri in fase di aggregazione.
  df <- read.csv(filepath, check.names = FALSE, stringsAsFactors = FALSE,
                 encoding = "UTF-8", colClasses = "character", na.strings = character(0))
  # File non UTF-8 (es. Latin-1 salvato da Excel)
  to_utf8 <- function(x) {
    bad <- !validUTF8(x)
    if (any(bad)) x[bad] <- iconv(x[bad], "latin1", "UTF-8")
    x
  }
  names(df) <- to_utf8(names(df))
  df[] <- lapply(df, to_utf8)
  # BOM UTF-8 all'inizio del file (byte EF BB BF, anche se letto come "ï»¿")
  b1 <- charToRaw(names(df)[1])
  for (bom in list(as.raw(c(0xef,0xbb,0xbf)), as.raw(c(0xc3,0xaf,0xc2,0xbb,0xc2,0xbf))))
    if (length(b1) > length(bom) && all(b1[seq_along(bom)] == bom))
      names(df)[1] <- rawToChar(b1[-seq_along(bom)])
  names(df) <- trimws(names(df))
  miss <- setdiff(c("unit","timestamp","treatment"), names(df))
  if (length(miss)) stop("ERR_COLS|", paste(miss, collapse = ", "), call. = FALSE)
  for (m in intersect(META_COLS, names(df))) df[[m]] <- trimws(df[[m]])
  for (m in c("unit","timestamp")) df[[m]][df[[m]] == "NA"] <- ""
  df$treatment[!nzchar(df$treatment)] <- NO_TREATMENT
  pd <- parse_pe_date(df$timestamp)
  df$date <- pd$date
  if (all(is.na(df$date))) stop("ERR_DATE", call. = FALSE)
  n_bad <- sum(is.na(df$date))
  df <- df[!is.na(df$date) & nzchar(df$unit), , drop = FALSE]
  parts  <- strsplit(df$unit, ":")
  getpos <- function(i) suppressWarnings(as.integer(vapply(parts,
              function(p) if (length(p) >= 3) p[i] else NA_character_, "")))
  df$posX <- getpos(2); df$posY <- getpos(3)
  if (!nrow(df) || any(is.na(df$posX) | is.na(df$posY))) stop("ERR_UNIT", call. = FALSE)
  df$posA <- vapply(parts, function(p) p[1], "")
  attr(df, "pe_notes") <- list(bad_dates = n_bad, ambiguous = pd$ambiguous)
  df
}

# ── Aggregazione giornaliera per pianta ───────────────────────────────────────
# Unità sperimentale = vaso (unit). Tutte le scansioni dello stesso vaso nello
# stesso giorno (ripetizioni nel giorno e, con più teste, i diversi sensori)
# sono repliche tecniche: si ottiene sempre UNA riga per pianta e giorno.
daily_mean_per_plant <- function(df, params) {
  params <- params[params %in% colnames(df)]
  key    <- paste(df$unit, format(df$date), sep = "\r")
  first  <- !duplicated(key)
  info   <- intersect(c("unit","date","treatment","posX","posY","genotype"), colnames(df))
  out    <- df[first, info, drop = FALSE]
  grp    <- factor(key, levels = key[first])
  out$n_scans <- as.integer(table(grp))
  for (p in params) {
    v <- suppressWarnings(as.numeric(df[[p]]))
    out[[p]] <- as.numeric(tapply(v, grp, function(x)
      if (all(is.na(x))) NA_real_ else mean(x, na.rm = TRUE)))
  }
  out <- out[order(out$date, out$posX, out$posY), , drop = FALSE]
  rownames(out) <- NULL
  out
}

# Valori presenti ma non numerici (es. decimale con la virgola), che diventano NA
count_non_numeric <- function(df, params) {
  params <- params[params %in% colnames(df)]
  n <- vapply(params, function(p) {
    s <- trimws(as.character(df[[p]]))
    sum(nzchar(s) & !(toupper(s) %in% c("NA","NAN")) &
          is.na(suppressWarnings(as.numeric(s))))
  }, 0L)
  n[n > 0]
}

# ── Aggregazione media±SE per trattamento × giorno ───────────────────────────
agg_trt_day <- function(daily_df, params, selected_units) {
  sub <- daily_df[daily_df$unit %in% selected_units, , drop = FALSE]
  if (!nrow(sub)) return(NULL)
  result <- list()
  for (p in params) {
    vals <- suppressWarnings(as.numeric(sub[[p]]))
    if (all(is.na(vals))) next
    tmp  <- sub; tmp$val <- vals
    m    <- aggregate(val ~ date + treatment, data = tmp, FUN = mean, na.rm = TRUE)
    s    <- aggregate(val ~ date + treatment, data = tmp, FUN = function(x) {
      n <- sum(!is.na(x)); if (n < 2) 0 else sd(x, na.rm=TRUE)/sqrt(n)
    })
    nn   <- aggregate(val ~ date + treatment, data = tmp, FUN = function(x) sum(!is.na(x)))
    names(m)[3] <- "mean"; names(s)[3] <- "se"; names(nn)[3] <- "n"
    merged <- merge(merge(m, s, by = c("date","treatment")), nn, by = c("date","treatment"))
    merged$param <- p
    result[[p]] <- merged
  }
  if (!length(result)) return(NULL)
  do.call(rbind, result)
}

# Testo leggibile sul vaso: scuro sui colori chiari (giallo, grigi chiari).
# I colori sono semitrasparenti, quindi conta anche lo sfondo della mappa.
label_color <- function(col, bg = "#FFFFFF") {
  m <- col2rgb(col, alpha = TRUE)[, 1] / 255
  v <- m[1:3] * m[4] + (col2rgb(bg)[, 1] / 255) * (1 - m[4])
  lin <- ifelse(v <= 0.03928, v / 12.92, ((v + 0.055) / 1.055)^2.4)
  if (sum(c(0.2126, 0.7152, 0.0722) * lin) > 0.45) "#1A1A1A" else "#FFFFFF"
}

      col  <- trt_colors[trt]

         </rect>',
        if (sel) " sel" else " excl",
        safe_id, x_svg, y_svg, cell_w-6, cell_h-6,
        rx, fill, strk,
        safe_id, uid, trt, if (sel) "1" else "0", lbl,
        uid, trt))
      # Mini etichetta unit
      cells <- c(cells, sprintf(

        safe_id, x_svg + (cell_w-6)/2, y_svg + (cell_h-6)/2 + 4,
        if (sel) lbl else "#9CA3AF",
        uid))
    }
    y_end <- y0 + n_row * cell_h
    y_top <- y_end + 16
  }
  # Legenda trattamenti: va a capo se non entra nella larghezza della mappa.

  max_chars <- max(4, floor((svg_w - 8 - swatch_w) / char_w))

                      paste0(substr(trts, 1, max_chars - 1), "\u2026"), trts)
  item_w    <- swatch_w + nchar(labels) * char_w + gap_w
  rows <- list(); cur <- integer(0); cur_w <- 0
  for (i in seq_along(trts)) {
    if (length(cur) && cur_w + item_w[i] - gap_w > svg_w - 8) {
      rows <- c(rows, list(cur)); cur <- integer(0); cur_w <- 0

# ── Grafico temporale ─────────────────────────────────────────────────────────
draw_time_chart <- function(trt_day_df, params, trt_colors, show_ribbon = TRUE,
                             ylab = "Mean value") {
  n_p   <- length(params)
  n_col <- 2; n_row <- ceiling(n_p / n_col)
  par(bg="#FFFFFF", col.axis="#1A1A1A", col.lab="#1A1A1A", col.main="#1A1A1A",
      fg="#1A1A1A", font.main=2, cex.main=1.35, cex.lab=1.2, cex.axis=1.1)
  if (n_p > 1) {
    par(mfrow=c(n_row, n_col), mar=c(7.5, 6, 3, 1.5), mgp=c(4, 0.9, 0))
  } else {
    par(mfrow=c(1,1), mar=c(7.5, 6, 3, 1.5), mgp=c(4, 0.9, 0))
  }

  trts      <- intersect(names(trt_colors), unique(trt_day_df$treatment))
  trts      <- trts[order(trts)]
  all_dates <- sort(unique(trt_day_df$date))
  date_ticks <- pretty(all_dates)
  date_ticks <- date_ticks[date_ticks >= min(all_dates) & date_ticks <= max(all_dates)]
  if (!length(date_ticks)) date_ticks <- all_dates

  for (p in params) {
    sub <- trt_day_df[trt_day_df$param == p, , drop=FALSE]
    if (!nrow(sub)) next

    if (!any(is.finite(sub$mean))) next
    # Taglio a 0 solo per variabili non negative (indici come PSRI possono essere < 0)
    nonneg <- all(sub$mean >= 0, na.rm=TRUE)
    ylim <- range(sub$mean + sub$se, sub$mean - sub$se, na.rm=TRUE)
    if (nonneg) ylim[1] <- max(0, ylim[1])
    if (diff(ylim) == 0) ylim <- ylim + c(-0.5, 0.5) * max(abs(ylim[1]), 1)
    ylim <- ylim + diff(ylim) * c(-0.05, 0.08)

    unit_m <- param_unit(p)

    plot.new()
    plot.window(xlim=range(all_dates), ylim=ylim)
    rect(par("usr")[1], par("usr")[3], par("usr")[2], par("usr")[4],
         col="#FFFFFF", border=NA)
    abline(h=pretty(ylim), col="#E5E5E5", lwd=0.9)
    abline(v=date_ticks, col="#F0F0F0", lwd=0.7)

    for (trt in trts) {
      d <- sub[sub$treatment==trt, ]
      if (!nrow(d)) next
      d <- d[order(d$date),]
      col <- trt_colors[trt]
      if (show_ribbon) {
        rib_col <- adjustcolor(col, alpha.f=0.18)
        polygon(c(d$date, rev(d$date)),
                c(d$mean + d$se, rev(if (nonneg) pmax(d$mean - d$se, 0) else d$mean - d$se)),
                col=rib_col, border=NA)
      }
      lines(d$date, d$mean, col=col, lwd=2.2)
      # Con un solo giorno non esiste una linea né una banda: punto e barra di SE
      if (nrow(d) == 1) {
        if (show_ribbon && isTRUE(d$se > 0)) {
          lo <- if (nonneg) max(d$mean - d$se, 0) else d$mean - d$se
          segments(d$date, lo, d$date, d$mean + d$se, col=col, lwd=2.2)
        }
        points(d$date, d$mean, col=col, pch=19, cex=1.3)
      }
    }

    axis(2, las=1, cex.axis=1.1, col="#1A1A1A", col.axis="#1A1A1A",
         mgp=c(4, 0.9, 0))
    box(col="#1A1A1A", lwd=1)

    # Etichette asse X ruotate di 30 gradi
    usr <- par("usr")
    y_off <- (usr[4] - usr[3]) * 0.03
    text(x=date_ticks, y=usr[3] - y_off,
         labels=format(date_ticks, "%d %b"),
         srt=30, adj=c(1,1), xpd=TRUE, cex=1.0, col="#1A1A1A")

    # Asse Y: nome parametro + unità
    ylab_full <- if (nchar(unit_m) > 0) paste0(ylab, " [", trimws(unit_m), "]") else ylab
    title(main=p, col.main="#1A1A1A")
    title(ylab=ylab_full, col.lab="#1A1A1A")

    # Legenda solo nel primo pannello, testo e simboli più grandi
    if (p == params[1]) {
      n_lab <- vapply(trts, function(tr) {
        nv <- trt_day_df$n[trt_day_df$treatment == tr & trt_day_df$param == p]
        if (!length(nv)) "" else if (min(nv) == max(nv)) paste0(" (n=", max(nv), ")")
        else paste0(" (n=", min(nv), "-", max(nv), ")")
      }, "")
      legend("topleft", legend=paste0(trts, n_lab), col=trt_colors[trts], lwd=2.5,
             bty="n", cex=1.05, pt.cex=1.5)
    }
  }
  par(mfrow=c(1,1))
}

draw_boxplot_day <- function(daily_df, params, trt_colors, selected_date,
                              ylab = "Value", show_jitter = FALSE) {
  sub <- daily_df[daily_df$date == selected_date, , drop = FALSE]
  if (!nrow(sub)) return(invisible(NULL))

  n_p   <- length(params)
  n_col <- 2; n_row <- ceiling(n_p / n_col)
  par(bg="#FFFFFF", col.axis="#1A1A1A", col.lab="#1A1A1A", col.main="#1A1A1A",
      fg="#1A1A1A", font.main=2, cex.main=1.3, cex.lab=1.2, cex.axis=1.1)
  if (n_p > 1) {
    par(mfrow=c(n_row, n_col), mar=c(7, 6, 3.5, 1.5), mgp=c(4, 0.9, 0))
  } else {
    par(mfrow=c(1,1), mar=c(7, 6, 3.5, 1.5), mgp=c(4, 0.9, 0))
  }
  trts <- intersect(names(trt_colors), unique(sub$treatment))
  trts <- trts[order(trts)]

  for (p in params) {
    vals <- suppressWarnings(as.numeric(sub[[p]]))
    sub$val <- vals
    bp_data <- lapply(trts, function(trt) {
      v <- sub$val[sub$treatment == trt]
      v[!is.na(v)]
    })
    names(bp_data) <- trts
    non_empty <- sapply(bp_data, length) > 0
    bp_data <- bp_data[non_empty]
    cols    <- trt_colors[names(bp_data)]

    unit_m   <- param_unit(p)
    ylab_full <- if (nchar(unit_m) > 0) paste0(ylab, " [", trimws(unit_m), "]") else ylab

    all_vals <- unlist(bp_data); all_vals <- all_vals[!is.na(all_vals)]
    if (!length(all_vals)) next
    ylim <- range(all_vals)
    pad  <- if (diff(ylim) > 0) diff(ylim) * 0.06 else max(abs(ylim[1]) * 0.05, 0.5)
    ylim <- ylim + c(-pad, pad)

    plot.new()
    plot.window(xlim=c(0.5, length(bp_data)+0.5), ylim=ylim)
    rect(par("usr")[1], par("usr")[3], par("usr")[2], par("usr")[4],
         col="#FFFFFF", border=NA)
    y_major <- pretty(ylim)
    y_major <- y_major[y_major >= ylim[1] & y_major <= ylim[2]]
    abline(h=y_major, col="#E5E5E5", lwd=0.9)

    bp_pos <- seq_along(bp_data)
    boxplot(bp_data, add=TRUE, at=bp_pos, col=cols,
            border="#1A1A1A", lwd=1, axes=FALSE,
            outline=TRUE, pch=19, cex=0.6,
            whisklty=1, staplelwd=1.2, boxwex=0.55)

    if (show_jitter) {
      for (j in seq_along(bp_data)) {
        vals_j <- bp_data[[j]]
        if (!length(vals_j)) next
        jx <- bp_pos[j] + runif(length(vals_j), -0.18, 0.18)
        points(jx, vals_j, pch=21,
               bg=adjustcolor("#1A1A1A", alpha.f=0.35),
               col=adjustcolor("#1A1A1A", alpha.f=0.55),
               cex=0.85, lwd=0.6)
      }
    }

    usr <- par("usr")
    y_off <- (usr[4] - usr[3]) * 0.04
    text(x=bp_pos, y=usr[3] - y_off,
         labels=paste0(names(bp_data), " (n=", lengths(bp_data), ")"),
         srt=30, adj=c(1,1), xpd=TRUE, cex=0.95, col="#1A1A1A")

    axis(2, at=y_major, col="#1A1A1A", col.axis="#1A1A1A", cex.axis=1.1,
         las=1, mgp=c(4,0.9,0))
    box(col="#1A1A1A", lwd=1)
    title(main=paste0(p, "  [", format(selected_date, "%d %b %Y"), "]"),
          col.main="#1A1A1A")
    title(ylab=ylab_full, col.lab="#1A1A1A")
  }
  par(mfrow=c(1,1))
}

# ── Pivot long ────────────────────────────────────────────────────────────────
pivot_long_pe <- function(df, params) {
  id_cols    <- c("unit","timestamp","date","treatment","sensor","genotype","posX","posY","n_scans")
  id_present <- id_cols[id_cols %in% colnames(df)]
  id_data    <- df[, id_present, drop=FALSE]
  long_rows  <- do.call(rbind, lapply(params, function(p) {
    tmp <- id_data; tmp$parameter <- p; tmp$value <- suppressWarnings(as.numeric(df[[p]])); tmp
  }))
  long_rows[order(rep(seq_len(nrow(df)), length(params))),]
}

# ── Base64 encoder ────────────────────────────────────────────────────────────
base64_encode_raw <- function(raw_bytes) {
  alphabet <- strsplit("ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/","")[[1]]
  n <- length(raw_bytes); pad <- (3-n%%3)%%3
  if (pad>0) raw_bytes <- c(raw_bytes, raw(pad))
  m   <- matrix(as.integer(raw_bytes), nrow=3)
  idx <- rbind(bitwShiftR(m[1,],2),
               bitwOr(bitwShiftL(bitwAnd(m[1,],3L),4), bitwShiftR(m[2,],4)),
               bitwOr(bitwShiftL(bitwAnd(m[2,],15L),2), bitwShiftR(m[3,],6)),
               bitwAnd(m[3,],63L))
  out <- matrix(alphabet[idx+1], nrow=4)
  if (pad>=1) out[4,ncol(out)] <- "="
  if (pad==2) out[3,ncol(out)] <- "="
  paste(out, collapse="")
}

# ── Serializza vettore R in array JS (per slider date) ───────────────────────
jsonlite_like_array_r <- function(x) {
  esc <- gsub('"', '\\\\"', as.character(x))
  paste0("[", paste0('"', esc, '"', collapse=","), "]")
}

csv_field <- function(x, sep) {
  x <- as.character(x); x[is.na(x)] <- ""
  need <- grepl(paste0("[", sep, "\"\r\n]"), x)
  x[need] <- paste0('"', gsub('"', '""', x[need], fixed = TRUE), '"')
  x
}
df_to_csv_string <- function(df, sep=",") {
  cols <- lapply(df, function(col) {
    if (inherits(col, "Date")) col <- format(col, "%Y-%m-%d")
    csv_field(col, sep)
  })
  header <- paste(csv_field(names(df), sep), collapse = sep)
  body   <- if (nrow(df)) do.call(paste, c(unname(cols), sep = sep)) else character(0)
  paste(c(header, body), collapse = "\n")
}

# Parametri da disegnare. NULL = nessuna scelta ancora fatta: solo i principali,
# perché i bin di distribuzione possono essere decine. character(0) = l'utente
# li ha deselezionati tutti, e quella scelta va rispettata.
resolve_chart_params <- function(chart_params, avail, main) {
  if (!is.null(chart_params)) return(intersect(avail, chart_params))
  m <- intersect(avail, main)
  if (length(m)) m else avail
}

server <- function(input, output, session) {
  rv <- reactiveValues(
    df_raw=NULL, df_daily=NULL, df_final=NULL,
    params=NULL, trt_colors=NULL, map_df=NULL,
    error=NULL, error_step=NULL, base_name=NULL, file_name=NULL,
    chart_png_b64=NULL, chart_png_box=NULL,

    # ridisegnati (cambio lingua, dark mode, rielaborazione)

    date_range=NULL, box_idx=NULL, multi_sensor=FALSE,
    load_notes=NULL, nonnum=NULL,
    is_example=FALSE, example_n=0L, palette=DEFAULT_PALETTE

  set_err <- function(step, msg) { rv$error <- msg; rv$error_step <- step }

  lang <- reactive({ if (is.null(input$ui_lang)) "en" else input$ui_lang })
  t    <- reactive({ TR[[lang()]] })

  observeEvent(input$selected_units, {
    new_units <- unlist(input$selected_units)
    # Il file finale già generato non rispecchierebbe più la mappa. La mappa
    # rimanda la selezione a ogni ridisegno: conta solo se è davvero cambiata.
    if (!setequal(new_units, rv$sel_units)) rv$df_final <- NULL
    rv$sel_units <- new_units
    if (length(rv$sel_units) && identical(rv$error, "ERR_NOSEL")) rv$error <- NULL
  }, ignoreNULL = FALSE, ignoreInit = TRUE)
  observeEvent(input$pivot_long, { rv$df_final <- NULL }, ignoreInit = TRUE)
  # La mappa cambia subito; i grafici già generati restano finché non li rigeneri
  observeEvent(input$palette, {
    rv$palette <- input$palette
    if (!is.null(rv$df_raw)) rv$trt_colors <- make_trt_colors(rv$df_raw$treatment, rv$palette)
  }, ignoreInit = TRUE)
  # Solo gli errori dello stesso passo: la selezione dello step 2 viene rimandata
  # anche al cambio lingua e non deve cancellare un errore dei grafici
  clear_param_err <- function(sel, steps) {
    if (length(sel) && isTRUE(rv$error %in% c("ERR_NOPARAM", "ERR_TOOMANY")) &&
        isTRUE(rv$error_step %in% steps)) rv$error <- NULL
  }
  observeEvent(input$selected_params, {
    rv$sel_params <- unlist(input$selected_params)
    clear_param_err(rv$sel_params, "clean")
  }, ignoreNULL = FALSE, ignoreInit = TRUE)
  observeEvent(input$chart_params, {
    # character(0), non NULL: "ho deselezionato tutto" è diverso da "non ho ancora scelto"
    rv$chart_params <- as.character(unlist(input$chart_params))
    clear_param_err(rv$chart_params, c("chart", "box"))
  }, ignoreNULL = FALSE, ignoreInit = TRUE)
  observeEvent(list(input$range_start_idx, input$range_end_idx), {
    if (!is.null(input$range_start_idx) && !is.null(input$range_end_idx))
      rv$date_range <- c(input$range_start_idx, input$range_end_idx)
  }, ignoreInit = TRUE)
  observeEvent(input$box_date_idx, { rv$box_idx <- input$box_date_idx }, ignoreInit = TRUE)

  for (st in c("load","clean","chart","box","final")) local({
    s <- st
    output[[paste0("err_", s)]] <- renderUI({
      req(rv$error, identical(rv$error_step, s))
      div(class="err-box", "\u26A0 ", tr_error(rv$error, t()))
    })
  })

  output$app_subtitle <- renderUI({
    tx <- t()
    tagList(
      p(class="app-subtitle", tx$app_subtitle, " ",
        tags$a(href="https://phenospex.com/products/plant-phenotyping/planteye-f600-multispectral-3d-scanner-for-plants/",
               target="_blank",
               style="color:#2D6A4F;font-weight:700;text-decoration:underline;",

  load_data <- function(path, file_name, is_example = FALSE) {
    rv$file_name <- file_name
    rv$is_example <- is_example
    rv$df_raw <- rv$df_daily <- rv$df_final <- rv$params <- rv$trt_colors <-
      rv$map_df <- rv$error <- rv$error_step <- rv$chart_png_b64 <- rv$chart_png_box <- NULL
    rv$sel_units <- rv$sel_params <- rv$chart_params <- rv$date_range <- rv$box_idx <- NULL
    rv$multi_sensor <- FALSE
    rv$load_notes <- rv$nonnum <- NULL
    tryCatch({
      df <- parse_planteye(path)
      p  <- classify_params(df)
      rv$base_name  <- sub("\\.csv$","", file_name, ignore.case=TRUE)
      rv$params     <- p
      rv$sel_params <- p$main
      rv$trt_colors <- make_trt_colors(df$treatment, rv$palette)
      rv$load_notes <- attr(df, "pe_notes")
      md <- unique(df[, c("unit","treatment","posA","posX","posY")])
      rv$map_df     <- md[!duplicated(md$unit), , drop=FALSE]
      rv$sel_units  <- rv$map_df$unit
      if ("sensor" %in% names(df)) {
        ns <- tapply(df$sensor, paste(df$unit, df$date), function(x) length(unique(x[nzchar(x)])))
        rv$multi_sensor <- any(ns > 1, na.rm=TRUE)
      }
      rv$df_raw <- df

    avail <- get_available_chart_params(df, c(rv$params$main, rv$params$dist))
    if (!length(avail)) return(NULL)
    act <- resolve_chart_params(isolate(rv$chart_params), avail, rv$params$main)
    rg  <- isolate(rv$date_range); if (length(rg) != 2) rg <- c(1, n)
    r0  <- max(1, min(as.integer(rg[1]), n)); r1 <- max(1, min(as.integer(rg[2]), n))
    rib <- isolate(input$show_ribbon); rib_on <- is.null(rib) || isTRUE(rib)

    pills <- lapply(avail, function(p)
      tags$span(class=paste("chart-param-toggle", if (p %in% rv$params$dist) "dist",
                            if (p %in% act) "active" else "excluded"),
                `data-param`=p, onclick="toggleChartParam(this);", p))

    div(class="sx-card",
      p(class="section-label", tx$step5),
      p(style="font-size:.9rem;color:#6B7280;margin-bottom:1rem;", tx$chart_hint),
      p(style="font-size:.9rem;color:#6B7280;font-weight:600;margin-bottom:.5rem;", tx$chart_param_lbl),
      div(class="chart-param-grid", pills),
      div(id="ribbon-toggle", class=if (rib_on) "pivot-toggle on" else "pivot-toggle", onclick="toggleRibbon();",
        div(class="pivot-switch"),
        div(div(class="pivot-label", tx$ribbon_label),
            div(class="pivot-sub",   tx$ribbon_sub))
      ),
      # ── Slicer date integrato ─────────────────────────────────────────────
      hr(class="section-divider", style="margin:.8rem 0;"),
      p(style="font-size:.9rem;color:#6B7280;margin-bottom:.8rem;", tx$range_hint),
      div(style="display:grid;grid-template-columns:1fr 1fr;gap:1.2rem;margin-bottom:.8rem;",
        div(
          tags$label(style="font-size:.88rem;font-weight:600;color:#6B7280;display:block;margin-bottom:.3rem;",
                     tx$range_start),
          tags$input(type="range", id="range_start_idx", min=1, max=n, value=r0, step=1,
                     style="width:100%;accent-color:#2D6A4F;", oninput="updateRangeLabel()")
        ),
        div(
          tags$label(style="font-size:.88rem;font-weight:600;color:#6B7280;display:block;margin-bottom:.3rem;",
                     tx$range_end),
          tags$input(type="range", id="range_end_idx", min=1, max=n, value=r1, step=1,
                     style="width:100%;accent-color:#2D6A4F;", oninput="updateRangeLabel()")
        )
      ),
      div(style="font-size:.92rem;color:#374151;margin-bottom:1rem;background:#F3F4F6;border-radius:6px;padding:.5rem 1rem;display:inline-block;",
        span(id="range_label", paste(format(dates[min(r0,r1)],"%d %b %Y"), "→", format(dates[max(r0,r1)],"%d %b %Y")))
      ),
      tags$script(HTML(sprintf("
        var _dates_range = %s;
        function updateRangeLabel() {
          var i0 = parseInt(document.getElementById('range_start_idx').value)-1;
          var i1 = parseInt(document.getElementById('range_end_idx').value)-1;
          if (i0 > i1) { var tmp=i0; i0=i1; i1=tmp; }
          document.getElementById('range_label').textContent =
            _dates_range[i0] + ' \u2192 ' + _dates_range[i1];
          Shiny.setInputValue('range_start_idx', i0+1, {priority:'event'});
          Shiny.setInputValue('range_end_idx',   i1+1, {priority:'event'});
        }
        updateRangeLabel();
      ", jsonlite_like_array_r(format(dates, "%d %b %Y"))))),
      # ── Pulsanti ─────────────────────────────────────────────────────────
      div(style="display:flex;gap:.8rem;align-items:center;flex-wrap:wrap;",
        actionButton("chart_btn", tx$btn_chart, icon=icon("chart-line"),
          style="border:none;color:#fff;font-weight:600;font-size:1rem;padding:.6rem 1.7rem;border-radius:6px;background:#0E7490;"),
        uiOutput("dl_png_ui"),
        uiOutput("clear_chart_ui")
      ),
      uiOutput("err_chart"),
      uiOutput("chart_render_wrap")
    )
  })

  observeEvent(input$chart_btn, {
    req(rv$df_daily, rv$trt_colors)
    rv$chart_png_b64 <- rv$error <- NULL
    sel_units <- rv$sel_units
    if (!length(sel_units)) { set_err("chart", "ERR_NOSEL"); return() }
    avail <- get_available_chart_params(rv$df_daily, c(rv$params$main, rv$params$dist))
    sel_params <- resolve_chart_params(rv$chart_params, avail, rv$params$main)
    sel_err <- chart_params_error(sel_params)
    if (!is.null(sel_err)) { set_err("chart", sel_err); return() }
    tryCatch({
      dates <- sort(unique(rv$df_daily$date))
      i0    <- max(1, min(as.integer(input$range_start_idx %||% 1),         length(dates)))
      i1    <- max(1, min(as.integer(input$range_end_idx   %||% length(dates)), length(dates)))
      if (i0 > i1) { tmp <- i0; i0 <- i1; i1 <- tmp }
      sub <- rv$df_daily[rv$df_daily$date %in% dates[i0:i1] & rv$df_daily$unit %in% sel_units,]
      trt_day <- agg_trt_day(sub, sel_params, unique(sub$unit))
      if (is.null(trt_day)||!nrow(trt_day)) stop("ERR_NODATA", call.=FALSE)
      sel_params <- sel_params[sel_params %in% trt_day$param]
      show_ribbon <- if (is.null(input$show_ribbon)) TRUE else isTRUE(input$show_ribbon)
      n_p <- length(sel_params); n_col <- if (n_p>1) 2 else 1; n_row <- ceiling(n_p/n_col)
      tmpf <- tempfile(fileext=".png")
      png(tmpf, width=5.5*n_col, height=5*n_row, units="in", res=300)
      draw_time_chart(trt_day, sel_params, rv$trt_colors, show_ribbon=show_ribbon, ylab=t()$chart_ylab)
      dev.off()
      raw_bytes <- readBin(tmpf, "raw", file.info(tmpf)$size)
      rv$chart_png_b64 <- base64_encode_raw(raw_bytes)
      unlink(tmpf)
    }, error=function(e) set_err("chart", conditionMessage(e)))
  })

  `%||%` <- function(a, b) if (is.null(a)) b else a

        tags$label(style="font-size:.88rem;font-weight:600;color:#6B7280;display:block;margin-bottom:.3rem;",

        tags$input(type="range", id="box_date_idx", min=1, max=n, value=mid, step=1,
                   style="width:100%;max-width:500px;accent-color:#7C3AED;",
                   oninput="updateBoxLabel()")
      ),
      div(style="font-size:.92rem;color:#374151;margin-bottom:1rem;background:#F3F4F6;border-radius:6px;padding:.5rem 1rem;display:inline-block;",
        span(id="box_label", format(dates[mid], "%d %b %Y"))
      ),
      tags$script(HTML(sprintf("
        var _dates_box = %s;
        function updateBoxLabel() {
          var i = parseInt(document.getElementById('box_date_idx').value)-1;
          document.getElementById('box_label').textContent = _dates_box[i];

        }
        updateBoxLabel();
      ", jsonlite_like_array_r(format(dates, "%d %b %Y"))))),
      div(id="jitter-box-toggle", class=if (jit_on) "pivot-toggle on" else "pivot-toggle", onclick="toggleJitterBox();",
        div(class="pivot-switch"),
        div(div(class="pivot-label", tx$jitter_label),
            div(class="pivot-sub",   tx$jitter_sub))
      ),
      div(style="display:flex;gap:.8rem;align-items:center;flex-wrap:wrap;",
        actionButton("box_chart_btn", tx$btn_box_chart, icon=icon("chart-bar"),
          style="border:none;color:#fff;font-weight:600;font-size:1rem;padding:.6rem 1.7rem;border-radius:6px;background:#7C3AED;"),
        uiOutput("dl_box_ui"),
        uiOutput("clear_box_ui")
      ),
      uiOutput("err_box"),
      uiOutput("box_chart_wrap")
    )
  })

      if (!length(sel_units)) stop("ERR_NOSEL", call.=FALSE)
      sub <- rv$df_daily[rv$df_daily$unit %in% sel_units, , drop=FALSE]
      avail <- get_available_chart_params(rv$df_daily, c(rv$params$main, rv$params$dist))
      sel_params <- resolve_chart_params(rv$chart_params, avail, rv$params$main)
      sel_err <- chart_params_error(sel_params)
      if (!is.null(sel_err)) stop(sel_err, call.=FALSE)
      sel_params <- sel_params[vapply(sel_params, function(p)
        any(!is.na(suppressWarnings(as.numeric(sub[[p]][sub$date == sel_date])))), logical(1))]
      if (!length(sel_params)) stop("ERR_NODATA", call.=FALSE)
      n_p <- length(sel_params); n_col <- if (n_p>1) 2 else 1; n_row <- ceiling(n_p/n_col)
      tmpf <- tempfile(fileext=".png")
      png(tmpf, width=5.5*n_col, height=5*n_row, units="in", res=300)
      draw_boxplot_day(sub, sel_params, rv$trt_colors, selected_date=sel_date,
                        ylab=t()$chart_ylab,
                        show_jitter=isTRUE(input$show_jitter_box))
      dev.off()
      raw_bytes <- readBin(tmpf, "raw", file.info(tmpf)$size)
      rv$chart_png_box <- base64_encode_raw(raw_bytes)
